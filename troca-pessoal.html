async function processarMatchPresencial(idColega) { document.getElementById('painelGerarQr').style.display = 'none'; const { data: perfilColega } = await _supabase.from('usuarios_cadastrados').select('nome').eq('id', idColega).maybeSingle(); document.getElementById('tituloMatchColega').innerText = `✨ Conferência: ${albumNomeSelecionado} com ${perfilColega?.nome || 'Colega'}`; if (!albumBaseSelecionadoId) { console.error("albumBaseSelecionadoId não está definido!"); return; } // 1. Busca global das figurinhas corrigida para usar 'album_id' (ajuste se a sua coluna tiver outro nome) const { data: figsDoAlbum, error: erroFigs } = await _supabase .from('figurinhas') .select('id, album_id, numero, nome, imagem') .eq('album_id', albumBaseSelecionadoId); if (erroFigs) { console.error("Erro ao buscar figurinhas:", erroFigs.message); return; } const idsFigurinhasAlbum = new Set((figsDoAlbum || []).map(f => String(f.id))); const figsMap = new Map((figsDoAlbum || []).map(f => [String(f.id), f])); // 2. Busca da cardeneta do usuário logado e do colega const [{ data: minhasFig }, { data: figColega }] = await Promise.all([ _supabase.from('usuario_cardeneta').select('figurinha_id, quantidade').eq('user_id', usuarioAtual.id), _supabase.from('usuario_cardeneta').select('figurinha_id, quantidade').eq('user_id', idColega) ]); const minhasDoAlbum = (minhasFig || []).filter(item => idsFigurinhasAlbum.has(String(item.figurinha_id))); const colegaDoAlbum = (figColega || []).filter(item => idsFigurinhasAlbum.has(String(item.figurinha_id))); const minhasCadastradasIds = new Set(minhasDoAlbum.map(item => String(item.figurinha_id))); const colecCadastradasIds = new Set(colegaDoAlbum.map(item => String(item.figurinha_id))); const minhasFaltantesIds = new Set(); idsFigurinhasAlbum.forEach(id => { if (!minhasCadastradasIds.has(id)) minhasFaltantesIds.add(id); }); const colegaRepetidas = colegaDoAlbum.filter(item => item.quantidade > 1); const minhasRepetidas = minhasDoAlbum.filter(item => item.quantidade > 1); // 3. Renderização dos grids de Receber e Enviar selecionadasParaReceber = new Set(); let htmlReceber = ''; colegaRepetidas.forEach(item => { const figId = String(item.figurinha_id); if (minhasFaltantesIds.has(figId)) { selecionadasParaReceber.add(figId); const f = figsMap.get(figId); if (f) { htmlReceber += `
${f.numero}
`; } } }); document.getElementById('gridReceberColega').innerHTML = htmlReceber || 'Nenhum match de recebimento para este álbum.'; selecionadasParaEnviar = new Set(); let htmlEnviar = ''; minhasRepetidas.forEach(item => { const figId = String(item.figurinha_id); if (!colecCadastradasIds.has(figId)) { selecionadasParaEnviar.add(figId); const f = figsMap.get(figId); if (f) { htmlEnviar += `
${f.numero}
`; } } }); document.getElementById('gridEnviarMim').innerHTML = htmlEnviar || 'Nenhum match de envio para este álbum.'; document.getElementById('painelMatchPresencial').style.display = 'block'; }
